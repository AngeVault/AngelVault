#pragma once
#include "ecs/behavior.h"
#include "ecs/registry.h"
#include "components/name.h"
#include "components/rigid_body2d.h"
#include <SFML/Window/Keyboard.hpp>

inline void shootBall(Registry& registry, int ballId, Entity self)
{
    auto& playerTrans = registry.getComponent<Transform2D>(self);
    auto& rbPlayer = registry.getComponent<RigidBody2D>(self);
    
    sf::Vector2f spawnPose = playerTrans.getPosition();
    spawnPose.x += 33; spawnPose.y += 1;

    Entity ball = registry.createEntity();
    registry.addComponent<Transform2D>(ball).setPosition(spawnPose);
    auto& rbBall = registry.addComponent<RigidBody2D>(ball);
    rbBall.mass = 0.5f;
    rbBall.applyImpulse({ 50 + rbPlayer.velocity.x * 2, 0 });
}

Module Réseau & HTTPS Asynchrone

Intégration de cURL et OpenSSL dans l'architecture du moteur.

http_client.h cURL / SSL 
#pragma once
#include <string>
#include <future>
#include <curl/curl.h>

namespace AngeVault {
struct HttpResponse {
    int statusCode = 0;
    std::string body;
    bool success = false;
};

class HttpClient {
public:
    static std::future<HttpResponse> getAsync(const std::string& url) {
        return std::async(std::launch::async, [url]() {
            HttpResponse res;
            CURL* curl = curl_easy_init();
            if (curl) {
                std::string buffer;
                curl_easy_setopt(curl, CURLOPT_URL, url.c_str());
                curl_easy_setopt(curl, CURLOPT_SSL_VERIFYPEER, 1L);
                CURLcode code = curl_easy_perform(curl);
                res.success = (code == CURLE_OK);
                curl_easy_cleanup(curl);
            }
            return res;
        });
    }
};
}
